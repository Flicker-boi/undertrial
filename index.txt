        function makeDualSwipeable(element, onDeleteCallback, onLeftSwipeCallback, onClickCallback) {
            let startX = 0;
            let currentTranslate = 0;
            let isDragging = false;

            const touchStart = (e) => {
                startX = e.type.includes('mouse') ? e.clientX : e.touches[0].clientX;
                isDragging = false;
                document.addEventListener('mousemove', touchMove);
                document.addEventListener('mouseup', touchEnd);
                document.addEventListener('touchmove', touchMove, { passive: true });
                document.addEventListener('touchend', touchEnd);
            };

            const touchMove = (e) => {
                const currentX = e.type.includes('mouse') ? e.clientX : e.touches[0].clientX;
                const diff = currentX - startX;
                if (Math.abs(diff) > 10) {
                    isDragging = true;
                    if (e.cancelable) e.preventDefault();
                }
                if (Math.abs(diff) < 180) {
                    currentTranslate = diff;
                    element.style.transform = `translateX(${currentTranslate}px)`;
                }
            };

            const touchEnd = () => {
                document.removeEventListener('mousemove', touchMove);
                document.removeEventListener('mouseup', touchEnd);
                document.removeEventListener('touchmove', touchMove);
                document.removeEventListener('touchend', touchEnd);

                if (currentTranslate > 100) {
                    onDeleteCallback();
                } else if (currentTranslate < -100) {
                    onLeftSwipeCallback();
                } else {
                    currentTranslate = 0;
                    element.style.transform = `translateX(0px)`;
                }
                setTimeout(() => { isDragging = false; }, 50);
            };

            element.addEventListener('mousedown', touchStart);
            element.addEventListener('touchstart', touchStart, { passive: true });

            if (onClickCallback) {
                element.addEventListener('click', (e) => {
                    if (!isDragging) onClickCallback(e);
                });
            }
        }

            const yanlis = parseInt(document.getElementById('yanlis').value) || 0;
            const bos = parseInt(document.getElementById('bos').value) || 0;

            if (!denemeAdi || !dersAdi || !konuAdi || !denemeTarihi || toplamSoru <= 0 || dersSoruSayisi <= 0 || denemeToplamSoru <= 0) {
                gosterFormUyarisi("* Lütfen tüm alanları eksiksiz doldurun, evladım!");

            let mevcutDersSoruToplami = 0;

            });

        let heartMaskImage = new Image();

        function drawPieChartWithHeartMask(canvasId, dogru, yanlis, bos) {
            const canvas = document.getElementById(canvasId);
            if (!canvas) return;
            const ctx = canvas.getContext('2d');
            ctx.clearRect(0, 0, canvas.width, canvas.height);

            let total = dogru + yanlis + bos;
            if (total <= 0) total = 1;

            let angles = [
                { count: dogru, color: '#00c000' },   
                { count: yanlis, color: '#ff0000' },  
                { count: bos, color: '#555555' }      
            ];

            let cx = canvas.width / 2;
            let cy = canvas.height / 2;
            let radius = Math.max(cx, cy) * 1.2;

            let tempCanvas = document.createElement('canvas');
            tempCanvas.width = canvas.width;
            tempCanvas.height = canvas.height;
            let tempCtx = tempCanvas.getContext('2d');

            let currentAngle = -Math.PI / 2;
            angles.forEach(slice => {
                if (slice.count <= 0) return;
                let sliceAngle = (slice.count / total) * (2 * Math.PI);
                tempCtx.beginPath();
                tempCtx.moveTo(cx, cy);
                tempCtx.arc(cx, cy, radius, currentAngle, currentAngle + sliceAngle + 0.05);
                tempCtx.closePath();
                tempCtx.fillStyle = slice.color;
                tempCtx.strokeStyle = slice.color;
                tempCtx.lineWidth = 2;
                tempCtx.fill();
                tempCtx.stroke();
                currentAngle += sliceAngle;
            });

            if (heartMaskImage.complete && heartMaskImage.naturalWidth !== 0) {
                tempCtx.globalCompositeOperation = 'destination-in';
                tempCtx.drawImage(heartMaskImage, 0, 0, tempCanvas.width, tempCanvas.height);
            }

            ctx.clearRect(0, 0, canvas.width, canvas.height);
            ctx.drawImage(tempCanvas, 0, 0);
        }

        function drawAnimatedGroupedBarChart(canvasId, dataList) {
            const canvas = document.getElementById(canvasId);
            if (!canvas) return;
            const ctx = canvas.getContext('2d');
            if (!dataList || dataList.length === 0) return;

            let dynamicWidth = Math.max(400, dataList.length * 60 + 50);
            canvas.width = dynamicWidth;
            canvas.style.width = dynamicWidth + "px";

            let maxSoru = 1;
            dataList.forEach(item => {
                let m = Math.max(item.dogru, item.yanlis, item.bos);
                if (m > maxSoru) maxSoru = m;
            });

            let paddingBottom = 25, paddingTop = 15;
            let chartHeight = canvas.height - paddingBottom - paddingTop;
            let chartWidth = canvas.width;
            let startTime = performance.now();
            let duration = 600;

            barHitboxes = [];

            function animate(currentTime) {
                let elapsed = currentTime - startTime;
                let progress = Math.min(elapsed / duration, 1);
                let easeProgress = 1 - Math.pow(1 - progress, 3);

                ctx.clearRect(0, 0, canvas.width, canvas.height);
                ctx.strokeStyle = '#333333';
                ctx.lineWidth = 1;
                for (let i = 0; i <= 4; i++) {
                    let y = paddingTop + (chartHeight / 4) * i;
                    ctx.beginPath(); ctx.moveTo(35, y); ctx.lineTo(chartWidth - 10, y); ctx.stroke();
                    let valLabel = Math.round(maxSoru - (maxSoru / 4) * i);
                    ctx.fillStyle = '#888888'; ctx.font = '9px sans-serif'; ctx.fillText(valLabel, 5, y + 3);
                }

                let groupWidth = (chartWidth - 45) / dataList.length;
                let barWidth = Math.max(5, Math.min(12, groupWidth / 4)); 
                let currentHitboxes = [];

                dataList.forEach((item, index) => {
                    let groupX = 45 + index * groupWidth + (groupWidth - (barWidth * 3 + 6)) / 2;
                    let hD = (item.dogru / maxSoru) * chartHeight * easeProgress;
                    let hY = (item.yanlis / maxSoru) * chartHeight * easeProgress;
                    let hB = (item.bos / maxSoru) * chartHeight * easeProgress;
                    let baseY = paddingTop + chartHeight;

                    let xD = groupX, xY = groupX + barWidth + 3, xB = groupX + (barWidth * 2) + 6;

                    ctx.fillStyle = '#00c000'; ctx.fillRect(xD, baseY - hD, barWidth, hD);
                    ctx.fillStyle = '#ff0000'; ctx.fillRect(xY, baseY - hY, barWidth, hY);
                    ctx.fillStyle = '#777777'; ctx.fillRect(xB, baseY - hB, barWidth, hB);

                    if (progress === 1) {
                        ctx.fillStyle = '#ffffff'; ctx.font = '8px sans-serif'; ctx.textAlign = 'center';
                        ctx.fillText(item.dogru, xD + barWidth/2, baseY - hD - 3);
                        ctx.fillText(item.yanlis, xY + barWidth/2, baseY - hY - 3);
                        ctx.fillText(item.bos, xB + barWidth/2, baseY - hB - 3);
                        ctx.textAlign = 'start';
                    }

                    ctx.fillStyle = '#ffffff'; ctx.font = '9px "WhFont", sans-serif'; ctx.textAlign = 'center';
                    let numX = groupX + barWidth + 3;
                    ctx.fillText(item.no, numX, canvas.height - 8);
                    ctx.textAlign = 'start';

                    if (progress === 1) {
                        currentHitboxes.push({
                            x: groupX - 6, y: 0, w: (barWidth * 3 + 6) + 12, h: canvas.height,
                            centerX: numX, topY: paddingTop,
                            text: `Deneme ${item.no} -> D: ${item.dogru} | Y: ${item.yanlis} | B: ${item.bos}`
                        });
                    }
                });

                barHitboxes = currentHitboxes;
                if (progress < 1) requestAnimationFrame(animate);
            }
            requestAnimationFrame(animate);
        }

        const barBoxEl = document.getElementById('barCanvasBox');

        barBoxEl.addEventListener('mousemove', (e) => {
            const rect = barBoxEl.getBoundingClientRect();
            const mouseX = e.clientX - rect.left + barBoxEl.scrollLeft;
            const mouseY = e.clientY - rect.top;
            let hovered = null;
            barHitboxes.items?.forEach ? null : barHitboxes.forEach(box => {
                if (mouseX >= box.x && mouseX <= box.x + box.w && mouseY >= box.y && mouseY <= box.y + box.h) hovered = box;
            });
            if (hovered) {
                tooltipEl.style.display = 'block';
                tooltipEl.innerText = hovered.text;
                tooltipEl.style.left = hovered.centerX + 'px';
                tooltipEl.style.top = (hovered.topY - 14) + 'px';
            } else {
                tooltipEl.style.display = 'none';
            }
        });

        barBoxEl.addEventListener('mouseleave', () => { tooltipEl.style.display = 'none'; });
